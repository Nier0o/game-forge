**Game Logic in Python**
========================

### Game Entities

We'll define simple entities for the player, enemies, and sword.

```python
import math

class Entity:
    def __init__(self, x, y):
        self.x = x
        self.y = y

class Player(Entity):
    def __init__(self, x, y):
        super().__init__(x, y)
        self.state = 'idle'
        self.sword_cooldown = 0

class Enemy(Entity):
    def __init__(self, x, y):
        super().__init__(x, y)
        self.state = 'idle'

class Sword:
    def __init__(self, owner):
        self.owner = owner
        self.cooldown = 0
```

### Game Logic

We'll create a simple game loop with update functions.

```python

    return math.sqrt((entity1.x - entity2.x) ** 2 + (entity1.y - entity2.y) ** 2)

            if distance(enemy, player) > 5:
                enemy.x += (player.x - enemy.x) / 10
                enemy.y += (player.y - enemy.y) / 10
            else:
                enemy.state = 'attack'

        elif enemy.state == 'attack':
            # Attack player if close enough
            if distance(enemy, player) <= 2:
                print("Enemy attacks player!")
            else:
                enemy.state = 'idle'

def update_sword(sword, player):

        sword.cooldown -= 1
```

### Game Loop

```python
def game_loop(player, enemies, sword):
    while True:
        handle_input(player, sword)
        update_player(player, enemies)
        update_enemies(enemies, player)
        update_sword(sword, player)

        # Print game state
        print(f"Player state: {player.state}, position: ({player.x}, {player.y})")
        for enemy in enemies:
            print(f"Enemy state: {enemy.state}, position: ({enemy.x}, {enemy.y})")

# Initialize game entities
player = Player(0, 0)
enemies = [Enemy(10, 10), Enemy(20, 20)]
sword = Sword(player)

# Start game loop
game_loop(player, enemies, sword)
```
